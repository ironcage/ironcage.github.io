Firebase 설정 복사

버튼을 누르면 복사돼요. Firebase 화면 글상자를 클릭 → Ctrl+A → Ctrl+V → 게시.

② Firestore 규칙

Firestore 규칙 화면 열기 ↗

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /hof/{id} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['name','nick','from','cls','style','rec','belts','def','score','first','high','f','at','dev'])
        && request.resource.data.name is string && request.resource.data.name.size() > 0 && request.resource.data.name.size() <= 24
        && request.resource.data.nick is string && request.resource.data.nick.size() <= 24
        && request.resource.data.score is int && request.resource.data.score >= 0 && request.resource.data.score <= 5000
        && request.resource.data.rec is list && request.resource.data.rec.size() == 3
        && request.resource.data.high is string && request.resource.data.high.size() <= 200
        && request.resource.data.f is string && request.resource.data.f.size() <= 8000
        && request.resource.data.dev is string && request.resource.data.dev.size() <= 40;
      allow update, delete: if false;
    }
    match /rank/{uid} {
      allow read: if true;
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['name','nick','cls','style','w','l','d','ko','sub','titles','def','score','champ','at'])
        && request.resource.data.name is string && request.resource.data.name.size() <= 24
        && request.resource.data.score is int && request.resource.data.score >= 0 && request.resource.data.score <= 5000
        && request.resource.data.w is int && request.resource.data.w >= 0 && request.resource.data.w <= 2000;
      allow delete: if request.auth != null && request.auth.uid == uid;
    }
    match /saves/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
  }
}

③ 실시간 데이터베이스 규칙

실시간 데이터베이스 화면 열기 ↗ (먼저 '데이터베이스 만들기' → 싱가포르 → 잠금 모드)

{
  "rules": {
    "ping": { ".read": "auth != null" },
    "rooms": {
      "$room": {
        ".read": "auth != null",
        "p": { "$uid": { ".write": "auth != null && auth.uid === $uid" } },
        "e": { ".write": "auth != null" }
      }
    }
  }
}
